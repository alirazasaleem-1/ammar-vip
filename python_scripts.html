Password Generator 🔐

🟢 Easy

Generate random strong passwords of any length. Never use "123456" again!

import random
import string

def generate_password(length=16):
    """Generate a random strong password."""
    characters = string.ascii_letters + string.digits + string.punctuation
    password = ''.join(random.choice(characters) for _ in range(length))
    return password

# Ask the user how long they want the password
print("🔐 Password Generator")
print("=" * 30)

length = int(input("How many characters? (default 16): ") or "16")

print(f"\n✅ Your password: {generate_password(length)}")
print(f"✅ Another option: {generate_password(length)}")
print(f"✅ One more:       {generate_password(length)}")

💡 What You'll Learn

  • • Importing modules (random, string)
  • • Functions with default parameters
  • • String joining and list comprehensions

Countdown Timer ⏱️

🟢 Easy

Set a timer and watch it count down in the terminal. Beeps when done!

import time
import sys

print("⏱️  Countdown Timer")
print("=" * 30)

seconds = int(input("Enter seconds to count down: "))

print(f"\nStarting countdown from {seconds} seconds...\n")

for i in range(seconds, 0, -1):
    minutes = i // 60
    secs = i % 60
    timer = f"{minutes:02d}:{secs:02d}"
    print(f"\r  ⏳ {timer} remaining...", end="")
    sys.stdout.flush()
    time.sleep(1)

print("\r  🔔 TIME'S UP!              ")
print("\a")  # Beep sound!
print("\n✅ Timer finished. Well done!")

💡 What You'll Learn

  • • time.sleep() for delays
  • • Carriage return \r for live updates
  • • Integer division and modulo

Dice Roller Game 🎲

Roll dice against the computer. Keep score and see who wins!

Bulk File Renamer 📁

🟡 Medium

Rename all files in a folder by adding a prefix or suffix. A real automation tool!

import os

print("📁 Bulk File Renamer")
print("=" * 30)

folder = input("Enter folder path: ").strip()

if not os.path.isdir(folder):
    print("❌ Folder not found!")
    exit()

print("\nOptions:")
print("  1. Add prefix (e.g., 'project_' + filename)")
print("  2. Add suffix (e.g., filename + '_backup')")

choice = input("\nChoose (1 or 2): ")
text = input("Enter the prefix/suffix text: ")

files = [f for f in os.listdir(folder) if os.path.isfile(os.path.join(folder, f))]
count = 0

for filename in files:
    name, ext = os.path.splitext(filename)
    
    if choice == "1":
        new_name = text + filename
    else:
        new_name = name + text + ext
    
    old_path = os.path.join(folder, filename)
    new_path = os.path.join(folder, new_name)
    os.rename(old_path, new_path)
    print(f"  ✅ {filename} → {new_name}")
    count += 1

print(f"\n🎉 Renamed {count} files!")

💡 What You'll Learn

  • • os module for file operations
  • • Path manipulation & splitting extensions
  • • Real-world file automation

Birthday Calculator 🎂

🟢 Easy

Enter your birthday and find out your exact age and days until your next birthday!

from datetime import datetime, date

print("🎂 Birthday Calculator")
print("=" * 30)

birth_str = input("Enter your birthday (DD-MM-YYYY): ")
birth_date = datetime.strptime(birth_str, "%d-%m-%Y").date()
today = date.today()

# Calculate age
age_years = today.year - birth_date.year
if (today.month, today.day) < (birth_date.month, birth_date.day):
    age_years -= 1

total_days = (today - birth_date).days

# Days until next birthday
next_bday = date(today.year, birth_date.month, birth_date.day)
if next_bday < today:
    next_bday = date(today.year + 1, birth_date.month, birth_date.day)
days_until = (next_bday - today).days

print(f"\n📊 Results:")
print(f"  🎈 You are {age_years} years old")
print(f"  📅 You've been alive for {total_days:,} days")
print(f"  ⏳ {days_until} days until your next birthday!")

if days_until == 0:
    print("  🎉 HAPPY BIRTHDAY!!!")

💡 What You'll Learn

  • • datetime module for dates
  • • Date parsing and arithmetic
  • • Tuple comparison

Rock Paper Scissors ✊✋✌️

🟢 Easy

Play the classic game against your computer. Best of 5 rounds!

import random

print("✊✋✌️ Rock Paper Scissors — Best of 5")
print("=" * 40)

choices = ["rock", "paper", "scissors"]
emojis = {"rock": "✊", "paper": "✋", "scissors": "✌️"}
wins = {"rock": "scissors", "paper": "rock", "scissors": "paper"}

you, cpu = 0, 0

while you < 3 and cpu < 3:
    pick = input("\nType rock, paper, or scissors: ").lower().strip()
    if pick not in choices:
        print("  ❌ Invalid choice, try again.")
        continue

    cpu_pick = random.choice(choices)
    print(f"  You: {emojis[pick]}  vs  Computer: {emojis[cpu_pick]}")

    if pick == cpu_pick:
        print("  🤝 Tie!")
    elif wins[pick] == cpu_pick:
        print("  ✅ You win!")
        you += 1
    else:
        print("  ❌ Computer wins!")
        cpu += 1
    print(f"  Score — You: {you} | Computer: {cpu}")

print(f"\n{'🏆 YOU WIN!' if you > cpu else '💻 Computer wins!'}")

💡 What You'll Learn

  • • Dictionaries for game logic
  • • While loops with conditions
  • • Input validation

Live Digital Clock 🕐

🟢 Easy

A live clock that updates every second in your terminal. Press Ctrl+C to stop.

import time
from datetime import datetime

print("🕐 Live Digital Clock (Press Ctrl+C to stop)\n")

try:
    while True:
        now = datetime.now()
        clock = now.strftime("%H:%M:%S")
        date = now.strftime("%A, %d %B %Y")
        
        print(f"\r  🕐  {clock}  |  📅 {date}  ", end="")
        time.sleep(1)

except KeyboardInterrupt:
    print("\n\n👋 Clock stopped. Goodbye!")

Folder Organizer 🗂️

🟡 Medium

Automatically sort files in any folder into subfolders by type. The king of automations!

import os
import shutil

CATEGORIES = {
    "Images": [".jpg", ".jpeg", ".png", ".gif", ".svg", ".webp"],
    "Documents": [".pdf", ".doc", ".docx", ".txt", ".xlsx", ".pptx"],
    "Videos": [".mp4", ".avi", ".mkv", ".mov"],
    "Music": [".mp3", ".wav", ".flac", ".aac"],
    "Code": [".py", ".js", ".html", ".css", ".java"],
    "Archives": [".zip", ".rar", ".7z", ".tar"],
}

def get_category(ext):
    for cat, exts in CATEGORIES.items():
        if ext.lower() in exts:
            return cat
    return "Other"

print("🗂️  Folder Organizer")
print("=" * 30)

folder = input("Enter folder path to organize: ").strip()

if not os.path.isdir(folder):
    print("❌ Folder not found!")
    exit()

count = 0
for filename in os.listdir(folder):
    filepath = os.path.join(folder, filename)
    if os.path.isfile(filepath):
        _, ext = os.path.splitext(filename)
        if not ext:
            continue
        
        category = get_category(ext)
        target_dir = os.path.join(folder, category)
        os.makedirs(target_dir, exist_ok=True)
        
        shutil.move(filepath, os.path.join(target_dir, filename))
        print(f"  ✅ {filename} → {category}/")
        count += 1

print(f"\n🎉 Organized {count} files!")

💡 What You'll Learn

  • • os + shutil for file management
  • • Creating directories dynamically
  • • Building real automation tools

Save as organizer.py
Run: python organizer.py
